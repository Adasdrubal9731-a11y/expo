import React, { useState } from "react";
import { View, Text, TextInput, TouchableOpacity, StyleSheet, ScrollView } from "react-native";

export default function App() {

const [screen, setScreen] = useState("home");
const [age, setAge] = useState("");
const [objectif, setObjectif] = useState("");
const [temps, setTemps] = useState("");
const [resultat, setResultat] = useState("");

const generateActivity = () => {
let suggestion = "";

if (objectif.toLowerCase().includes("sport")) {
suggestion = "🏃 Activité : marche rapide ou séance sport fun de 20 min.";
}
else if (objectif.toLowerCase().includes("cuisine")) {
suggestion = "🍰 Activité : essaie une nouvelle recette aujourd’hui.";
}
else if (objectif.toLowerCase().includes("culture")) {
suggestion = "📚 Découverte : apprends 5 faits culturels sur un pays.";
}
else {
suggestion = "🎨 Activité créative : dessin, écriture ou DIY relaxant.";
}

setResultat(suggestion);
setScreen("result");
};

return (
<View style={styles.container}>

{screen === "home" && (
<>
<Text style={styles.logo}>🌸 BLOOM</Text>
<Text style={styles.subtitle}>Grandis chaque jour</Text>

<TouchableOpacity style={styles.button} onPress={() => setScreen("question1")}>
<Text style={styles.buttonText}>Commencer</Text>
</TouchableOpacity>
</>
)}

{screen === "question1" && (
<>
<Text style={styles.question}>Quel est ton âge ?</Text>
<TextInput style={styles.input} value={age} onChangeText={setAge} keyboardType="numeric" />
<TouchableOpacity style={styles.button} onPress={() => setScreen("question2")}>
<Text style={styles.buttonText}>Suivant</Text>
</TouchableOpacity>
</>
)}

{screen === "question2" && (
<>
<Text style={styles.question}>Ton objectif principal ?</Text>
<TextInput 
style={styles.input} 
placeholder="sport, culture, cuisine, détente…" 
value={objectif} 
onChangeText={setObjectif}
/>
<TouchableOpacity style={styles.button} onPress={() => setScreen("question3")}>
<Text style={styles.buttonText}>Suivant</Text>
</TouchableOpacity>
</>
)}

{screen === "question3" && (
<>
<Text style={styles.question}>Temps libre par jour ?</Text>
<TextInput 
style={styles.input} 
placeholder="10 min, 1h…" 
value={temps} 
onChangeText={setTemps}
/>

<TouchableOpacity style={styles.button} onPress={generateActivity}>
<Text style={styles.buttonText}>Voir mon activité</Text>
</TouchableOpacity>
</>
)}

{screen === "result" && (
<ScrollView>
<Text style={styles.resultTitle}>✨ Ton activité du jour</Text>
<Text style={styles.result}>{resultat}</Text>

<Text style={styles.tip}>
💡 Conseil BLOOM : La régularité te fait évoluer. Reviens demain pour une nouvelle activité.
</Text>

<TouchableOpacity style={styles.button} onPress={() => setScreen("home")}>
<Text style={styles.buttonText}>Recommencer</Text>
</TouchableOpacity>
</ScrollView>
)}

</View>
);
}

const styles = StyleSheet.create({
container: {
flex: 1,
backgroundColor: "#ff2f92",
alignItems: "center",
justifyContent: "center",
padding: 20,
},

logo: {
fontSize: 42,
color: "white",
fontWeight: "bold",
marginBottom: 10,
},

subtitle: {
fontSize: 18,
color: "white",
marginBottom: 40,
},

question: {
fontSize: 22,
color: "white",
marginBottom: 20,
textAlign: "center"
},

input: {
backgroundColor: "white",
width: 250,
padding: 12,
borderRadius: 12,
marginBottom: 20,
},

button: {
backgroundColor: "#ffe600",
padding: 15,
borderRadius: 20,
marginTop: 10,
},

buttonText: {
fontWeight: "bold",
fontSize: 16,
},

resultTitle: {
fontSize: 26,
color: "white",
textAlign: "center",
marginBottom: 20,
},

result: {
fontSize: 20,
color: "white",
textAlign: "center",
marginBottom: 30,
},

tip: {
fontSize: 16,
color: "white",
textAlign: "center",
marginBottom: 40,
}
});
